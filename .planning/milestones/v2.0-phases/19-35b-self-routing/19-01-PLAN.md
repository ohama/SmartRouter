**Create `src/SmartRouter.Cli/Adapters/SelfRouter.fs`** as a near-line-by-line port of `src/SmartRouter.Cli/Adapters/JudgeClient.fs` (RESEARCH §1 confirms this is the canonical template). Use the following structure (mirroring JudgeClient sections by line range):

```fsharp
namespace SmartRouter.Cli.Adapters

module SelfRouter =

    open System
    open System.IO
    open System.Net.Http
    open System.Security.Cryptography
    open System.Text
    open System.Text.Json
    open System.Threading
    open System.Threading.Tasks
    open System.Collections.Concurrent
    open System.Collections.Generic
    open Microsoft.Extensions.Logging
    open Serilog
    open SmartRouter.Core
    open SmartRouter.Cli.Adapters.DecisionLogger   // for computePromptHash

    // --- 1. Verdict DU (SR-03) ---
    type SelfRouteVerdict =
        | RouteSafe
        | RouteUnsafe
        | RouteSkipped of reason: string
        | RouteFailed  of error: string

    // --- 2. CLIMutable options record (consumed by 19-02 DI; defaults in code, override via Routing:SelfRouter:* config) ---
    [<CLIMutable>]
    type SelfRouterOptions = {
        mutable Endpoint        : string   // empty → DI resolves to Upstreams.Model35B
        mutable PromptPath      : string   // default "prompts/self-router-prompt.md"
        mutable TimeoutSeconds  : int      // default 5 (NOT 300 — classify must be quick)
        mutable MaxCacheEntries : int      // default 10000
    }

    // --- 3. Port interface (SR-01) ---
    type ISelfRouter =
        abstract member ClassifyAsync :
            promptHash: string * promptText: string * ct: CancellationToken
            -> Task<SelfRouteVerdict>
        abstract member PromptVersion : string   // SHA-256 prefix of prompt file; stamped as model_version (SR-01 / SC-1)

    // --- 4. Stats interface (SR-05) ---
    type ISelfRouterStats =
        abstract member GetSelfRouterStats : unit -> struct (int64 * int64 * int64 * int64)
        // tuple: (cacheHits, cacheMisses, callCount, skipped)

    // --- 5. Cache entry with mutable AccessSeq for LRU eviction (mirrors JudgeClient.CacheEntry) ---
    type private CacheEntry = { Verdict: SelfRouteVerdict; mutable AccessSeq: int64 }

    // --- 6. SAFETY-BIASED parser (SR-03; PITFALL #1 in RESEARCH §9 — UNSAFE check FIRST) ---
    let private parseContent (content: string) : SelfRouteVerdict =
        let hasUnsafe = content.Contains("UNSAFE", StringComparison.OrdinalIgnoreCase)
        let hasSafe   = content.Contains("SAFE",   StringComparison.OrdinalIgnoreCase)
        match hasUnsafe, hasSafe with
        | true,  _    -> RouteUnsafe         // UNSAFE wins on collision (SAFE ⊂ UNSAFE — load-bearing order)
        | false, true -> RouteSafe
        | false, false ->
            let snippet = content.Substring(0, min 100 content.Length)
            RouteFailed (sprintf "unparseable self-router response: %s" snippet)

    // --- 7. Response envelope extractor (mlx_lm.server returns OpenAI-shape choices[0].message.content) ---
    let private tryReadResponseContent (json: string) : Result<string, string> =
        // Mirror JudgeClient.tryReadResponseContent exactly:
        //   parse JsonDocument, navigate "choices"[0]."message"."content" string,
        //   return Ok content on success, Error reason on any null/missing/wrong-type.

    // --- 8. Concrete class implementing ISelfRouter + ISelfRouterStats ---
    type SelfRouter
            (factory: IHttpClientFactory,
             opts:    SelfRouterOptions,
             logger:  ILogger<SelfRouter>) =

        // mutable interlocked counters (SR-05); use `mutable` field + Interlocked.Increment
        let mutable cacheHits   = 0L
        let mutable cacheMisses = 0L
        let mutable callCount   = 0L
        let mutable skippedCnt  = 0L
        let mutable accessSeq   = 0L  // monotonic LRU counter (Interlocked.Increment per touch)

        let cache       = ConcurrentDictionary<string, CacheEntry>()
        let maxEntries  = if opts.MaxCacheEntries > 0 then opts.MaxCacheEntries else 10000
        let timeoutSec  = if opts.TimeoutSeconds  > 0 then opts.TimeoutSeconds  else 5
        let promptPath  = if String.IsNullOrWhiteSpace(opts.PromptPath) then "prompts/self-router-prompt.md" else opts.PromptPath

        // --- Prompt template loaded on first ClassifyAsync call (lock-on-first-read; mirrors JudgeClient.getPromptTemplate) ---
        let templateLock = obj()
        let mutable templateCache : string option = None
        let getPromptTemplate () : string option =
            match templateCache with
            | Some s -> if s = "" then None else Some s
            | None ->
                lock templateLock (fun () ->
                    match templateCache with
                    | Some s -> if s = "" then None else Some s
                    | None ->
                        if File.Exists(promptPath) then
                            let content = File.ReadAllText(promptPath)
                            templateCache <- Some content
                            Some content
                        else
                            templateCache <- Some ""   // cache miss so we don't re-stat every call
                            None)

        // --- PromptVersion (RESEARCH §3, Open Question #2): SHA-256 of prompt file at construction time ---
        let promptVersion : string =
            try
                if File.Exists(promptPath) then
                    use sha = SHA256.Create()
                    let bytes = File.ReadAllBytes(promptPath)
                    let hash  = sha.ComputeHash(bytes)
                    let hex   = hash |> Array.map (sprintf "%02x") |> String.concat ""
                    sprintf "selfrouting-%s" (hex.Substring(0, 8))
                else
                    "selfrouting-v1"    // fallback when prompt file absent (operator must add it)
            with _ -> "selfrouting-v1"

        // --- LRU cache helpers (mirrors JudgeClient.tryGetCached/setCached, single-string key) ---
        let tryGetCached (key: string) : SelfRouteVerdict option =
            match cache.TryGetValue(key) with
            | true, entry ->
                entry.AccessSeq <- Interlocked.Increment(&accessSeq)
                Some entry.Verdict
            | false, _ -> None

        let setCached (key: string) (verdict: SelfRouteVerdict) : unit =
            // Bound check + LRU eviction (TOCTOU note: two threads may both evict — acceptable, mirrors JudgeClient lines 205-208)
            if cache.Count >= maxEntries then
                let victim =
                    cache
                    |> Seq.minBy (fun kv -> kv.Value.AccessSeq)
                cache.TryRemove(victim.Key) |> ignore
            let entry = { Verdict = verdict; AccessSeq = Interlocked.Increment(&accessSeq) }
            cache.[key] <- entry

        // --- buildBody (PITFALL §1 in RESEARCH §1: JsonFSharpConverter REQUIRED) ---
        let buildBody (promptText: string) : string =
            let template =
                match getPromptTemplate () with
                | Some t -> t
                | None   -> "Classify the following request as SAFE or UNSAFE for a small model. Respond with only the word SAFE or UNSAFE.\n\nRequest:\n{{PROMPT}}"
            let filled = template.Replace("{{PROMPT}}", promptText)
            let bodyDict = Dictionary<string, obj>()
            bodyDict.["model"]       <- "qwen-35b"   // logical name; mlx_lm.server resolves
            bodyDict.["messages"]    <- [| {| role = "user"; content = filled |} |]
            bodyDict.["max_tokens"]  <- 8            // SR-02; researcher recommends 8 over 4 for whitespace/punctuation absorption
            bodyDict.["temperature"] <- 0.0
            bodyDict.["stream"]      <- false
            // JsonFSharpConverter REQUIRED — anonymous records in messages array do NOT serialize without it.
            // This burned Phase 16; preserve it or this adapter will silently misformat the body.
            let optsJson = JsonSerializerOptions()
            optsJson.Converters.Add(System.Text.Json.Serialization.JsonFSharpConverter())
            JsonSerializer.Serialize(bodyDict, optsJson)

        // --- attemptOnce: single HTTP call with task {} (NO async {} — ARCH-02) ---
        let attemptOnce (body: string) (ct: CancellationToken) : Task<Result<string, string>> = task {
            let client = factory.CreateClient("selfrouter")
            // BaseAddress + Timeout set by DI (Plan 19-02); do NOT override here.
            use content = new StringContent(body, Encoding.UTF8, "application/json")
            try
                use! response = client.PostAsync("/v1/chat/completions", content, ct)
                if response.IsSuccessStatusCode then
                    let! json = response.Content.ReadAsStringAsync(ct)
                    return tryReadResponseContent json
                else
                    return Error (sprintf "selfrouter HTTP %d" (int response.StatusCode))
            with
            | :? OperationCanceledException -> return Error "selfrouter cancelled"
            | :? HttpRequestException as ex -> return Error (sprintf "selfrouter http: %s" ex.Message)
            | :? TaskCanceledException as ex -> return Error (sprintf "selfrouter timeout: %s" ex.Message)
            | ex -> return Error (sprintf "selfrouter unexpected: %s" ex.Message)
        }

        interface ISelfRouter with
            member _.PromptVersion = promptVersion

            member _.ClassifyAsync(promptHash, promptText, ct) = task {
                // 1) Cache check
                match tryGetCached promptHash with
                | Some verdict ->
                    Interlocked.Increment(&cacheHits) |> ignore
                    return verdict
                | None ->
                    Interlocked.Increment(&cacheMisses) |> ignore

                    // 2) Template availability check (missing template → RouteSkipped, NOT cached)
                    match getPromptTemplate () with
                    | None ->
                        Interlocked.Increment(&skippedCnt) |> ignore
                        return RouteSkipped (sprintf "self-router prompt template missing at %s" promptPath)
                    | Some _ ->
                        // 3) HTTP call
                        Interlocked.Increment(&callCount) |> ignore
                        let body = buildBody promptText
                        let! result = attemptOnce body ct
                        match result with
                        | Ok content ->
                            let verdict = parseContent content
                            // 4) Cache only RouteSafe / RouteUnsafe (PITFALL #4: never cache RouteFailed)
                            match verdict with
                            | RouteSafe | RouteUnsafe -> setCached promptHash verdict
                            | _ -> ()
                            return verdict
                        | Error e ->
                            return RouteFailed e
            }

        interface ISelfRouterStats with
            member _.GetSelfRouterStats() =
                struct (Volatile.Read(&cacheHits),
                        Volatile.Read(&cacheMisses),
                        Volatile.Read(&callCount),
                        Volatile.Read(&skippedCnt))
```

You are a routing classifier.

Your task is to determine whether a request is SAFE for a fast 35B coding model.

A request is SAFE if it requires only:

- formatting / summaries / boilerplate generation
- simple explanations
- basic code snippets

Do not explain. Do not add punctuation.

Request:
{{PROMPT}}

**Critical requirements** (verified by parser + integration tests in 19-03):

4. **No trailing newline beyond the standard final-newline** — preserve clean replacement.
