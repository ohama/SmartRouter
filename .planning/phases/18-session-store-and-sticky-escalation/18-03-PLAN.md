**1. ExecuteAsync pattern — PeriodicTimer + OCE catch (RESOLVED).**

Per 18-RESEARCH Pattern 7 and the Phase 8 RetrainingService precedent: use `System.Threading.PeriodicTimer` with 5-minute interval. On each tick, scan `store` and remove entries where `LastAccessedAt < cutoff`. Catch `OperationCanceledException` to exit cleanly on shutdown. Log non-OCE exceptions but DO NOT exit the loop — log-and-continue per the Phase 8 pattern.

```fsharp
override _.ExecuteAsync(ct: CancellationToken) : Task =
    task {
        use timer = new PeriodicTimer(TimeSpan.FromMinutes(5.0))
        let mutable go = true
        while go && not ct.IsCancellationRequested do
            try
                let! _ = timer.WaitForNextTickAsync(ct)
                let cutoff = DateTimeOffset.UtcNow.AddMinutes(float -ttlMinutes)
                for kv in store do
                    if kv.Value.LastAccessedAt < cutoff then
                        store.TryRemove(kv.Key) |> ignore
            with
            | :? OperationCanceledException -> go <- false
            | ex -> logger.LogError(ex, "SessionStore TTL eviction error")
    }
```

The simpler `:? OperationCanceledException -> go <- false` catch (vs the Phase 8 `ExceptionDispatchInfo.Capture` pattern) is sufficient here because there are no follow-up `do!` awaits after the `let! _ = timer.WaitForNextTickAsync(ct)` that would need OCE re-throw — the `for kv in store` block is synchronous. (`ExceptionDispatchInfo.Capture` is only required when re-throwing OCE through a chain of awaits to preserve the original cancellation token reference; here we don't need to re-throw.)

    ```fsharp
    /// Phase 18 (SES-08) — TTL eviction loop.
    /// PeriodicTimer 5-minute interval (hardcoded; SES-08); on each tick, scan the
    /// store and remove entries with LastAccessedAt older than ttlMinutes.
    ///
    /// Mirrors Phase 8 RetrainingService PeriodicTimer pattern: catch
    /// OperationCanceledException to exit cleanly on shutdown; log non-OCE exceptions
    /// but log-and-continue (do not exit the loop on transient errors).
    ///
    /// The simpler ":? OperationCanceledException -> go <- false" catch is sufficient
    /// here because there are no follow-up do! awaits after WaitForNextTickAsync —
    /// the eviction `for kv in store` block is synchronous. ExceptionDispatchInfo.Capture
    /// pattern (Phase 8) is only required when re-throwing OCE through a chain of awaits.
    override _.ExecuteAsync(ct: CancellationToken) : Task =
        task {
            logger.LogInformation(
                "SessionStore TTL eviction started; interval=5min ttlMinutes={Ttl}",
                ttlMinutes)
            use timer = new PeriodicTimer(TimeSpan.FromMinutes(5.0))
            let mutable go = true
            while go && not ct.IsCancellationRequested do
                try
                    let! _ = timer.WaitForNextTickAsync(ct)
                    let cutoff = DateTimeOffset.UtcNow.AddMinutes(float -ttlMinutes)
                    let mutable removed = 0
                    for kv in store do
                        if kv.Value.LastAccessedAt < cutoff then
                            if store.TryRemove(kv.Key) |> fst then
                                removed <- removed + 1
                    if removed > 0 then
                        logger.LogDebug(
                            "SessionStore TTL eviction removed {Removed} entries; remaining={Count}",
                            removed, store.Count)
                with
                | :? OperationCanceledException -> go <- false
                | ex ->
                    logger.LogError(ex, "SessionStore TTL eviction error; loop continues")
        }
    ```

    Note: `store.TryRemove(key) |> fst` checks the boolean part of the `(bool * 'a)` tuple to confirm removal happened (returns `false` if a concurrent thread already removed it). The dictionary enumeration via `for kv in store` is safe under concurrent mutation per `ConcurrentDictionary`'s documented enumeration semantics (snapshot-like; may miss recent insertions, no exception).

    ```fsharp
    module SmartRouter.Tests.SessionStoreTests

    open System
    open System.Threading
    open System.Threading.Tasks
    open Expecto
    open Microsoft.Extensions.Logging.Abstractions
    open SmartRouter.Core.Domain
    open SmartRouter.Cli.Adapters.SessionStore

    /// Helper: construct a SessionStore with low TTL + small LRU cap to make
    /// eviction visible inside a single test method execution.
    let private mkStore (ttlMinutes: int) (maxEntries: int) : SessionStore * ISessionStore =
        let opts = { TtlMinutes = ttlMinutes; MaxEntries = maxEntries }
        let store = SessionStore(opts, NullLogger<SessionStore>.Instance)
        store, (store :> ISessionStore)

    let tests : Test =
        testList "SessionStoreTests" [

            // ── Empty-sessionId no-op (Pitfall 7 prevention) ──

            testCase "Update with empty sessionId is no-op" <| fun () ->
                let store, iface = mkStore 30 1000
                iface.Update("", Qwen122B)
                Expect.equal store.Store.Count 0 "no entry created for empty sessionId"

            testCase "TryGet with empty sessionId returns None" <| fun () ->
                let _, iface = mkStore 30 1000
                Expect.isNone (iface.TryGet("")) "empty sessionId → None"

            // ── Basic write+read ──

            testCase "Update then TryGet returns the written state" <| fun () ->
                let _, iface = mkStore 30 1000
                iface.Update("sess-1", Qwen35B)
                let r = iface.TryGet("sess-1")
                Expect.isSome r "entry must exist"
                Expect.equal r.Value.LastModel Qwen35B "model recorded"

            testCase "TryGet on unknown sessionId returns None" <| fun () ->
                let _, iface = mkStore 30 1000
                iface.Update("sess-x", Qwen35B)
                Expect.isNone (iface.TryGet("sess-y")) "unknown session → None"

            // ── 122B-wins concurrent merge (Pitfall 4 — non-negotiable correctness) ──

            testCase "Concurrent Update: 122B wins over 35B regardless of race order" <| fun () ->
                let store, iface = mkStore 30 1000
                // Fire 100 35B writes + 100 122B writes concurrently against the same session.
                let writes =
                    [| for i in 1 .. 100 ->
                        Task.Run(fun () -> iface.Update("race", Qwen35B))
                       for i in 1 .. 100 ->
                        Task.Run(fun () -> iface.Update("race", Qwen122B)) |]
                Task.WaitAll(writes)
                let r = iface.TryGet("race")
                Expect.isSome r "entry must exist"
                Expect.equal r.Value.LastModel Qwen122B "122B-wins merge: final LastModel must be Qwen122B"

            testCase "Update 122B then 35B keeps 122B (sequential merge)" <| fun () ->
                let _, iface = mkStore 30 1000
                iface.Update("sess-merge", Qwen122B)
                iface.Update("sess-merge", Qwen35B)
                let r = iface.TryGet("sess-merge")
                Expect.equal r.Value.LastModel Qwen122B "122B stays even when newer write is 35B"

            // ── TTL eviction (SES-08 unit) ──

            testCase "TryGet returns None for entries older than TtlMinutes" <| fun () ->
                // TTL = 0 minutes → any entry is immediately stale.
                // SessionStore class normalizes <= 0 to 30; use TTL=1 + sleep to be deterministic.

                let store, iface = mkStore 30 1000

                match store.Store.TryGetValue("sess-stale") with
                | true, entry ->
                    let stale = { entry with LastAccessedAt = DateTimeOffset.UtcNow.AddMinutes(-60.0) }
                    store.Store.[("sess-stale")] <- stale
                    let r = iface.TryGet("sess-stale")
                    Expect.isNone r "TTL-expired entry → None"
                | _ ->
                    failtest "test setup failed: entry not created"

            testCase "MaxEntries cap evicts oldest-AccessSeq entry on overflow" <| fun () ->
    ```
