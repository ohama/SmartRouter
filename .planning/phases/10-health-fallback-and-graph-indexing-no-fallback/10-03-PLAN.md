let startFakeUpstream (respond : HttpContext -> Task<unit>) =
    let builder = WebApplication.CreateBuilder()
    builder.WebHost.UseUrls("http://127.0.0.1:0") |> ignore
    builder.Logging.ClearProviders() |> ignore
    let app = builder.Build()
    app.Run(fun ctx -> respond ctx) |> ignore
    app.StartAsync().GetAwaiter().GetResult()
    let port =
        let server = app.Services.GetRequiredService<Microsoft.AspNetCore.Hosting.Server.IServer>()
        let feat = server.Features.Get<Microsoft.AspNetCore.Hosting.Server.Features.IServerAddressesFeature>()
        let addr = Seq.head feat.Addresses
        Uri(addr).Port
    port, { new IDisposable with member _.Dispose() = app.StopAsync().GetAwaiter().GetResult(); app.DisposeAsync().GetAwaiter().GetResult() }

let private parseDecisionLog (logsDir: string) : (Map<string, JsonElement>) list =
    let today = DateTimeOffset.UtcNow.ToString("yyyy-MM-dd")
    let path = Path.Combine(logsDir, today + ".jsonl")
    if not (File.Exists(path)) then []
    else
        File.ReadAllLines(path)
        |> Array.filter (fun l -> not (String.IsNullOrWhiteSpace(l)))
        |> Array.map (fun line ->
            let doc = JsonDocument.Parse(line)
            doc.RootElement.EnumerateObject()
            |> Seq.map (fun p -> p.Name, p.Value.Clone())
            |> Map.ofSeq)
        |> Array.toList

/// Wait until predicate is true or timeout. Polls every 200ms.
let private waitFor (timeoutMs: int) (pred: unit -> bool) =
    let sw = Diagnostics.Stopwatch.StartNew()
    while sw.ElapsedMilliseconds < int64 timeoutMs && not (pred ()) do
        Thread.Sleep(200)
    pred ()

**Important implementation notes:**

1. **`startTestRouter` reuse.** The skeleton above has a `failwith` placeholder. **Read `tests/SmartRouter.Tests/StreamingTests.fs`** for the actual pattern (look for `let startTestRouter` or `let inline startRouter` near the top), and adapt to accept the additional config keys (`Routing:Health:*`). If StreamingTests' helper does not parameterize `Upstreams:Model122B`, extend it.

2. **`Routing:Algorithm = heuristic`.** Avoid the ML.zip dependency for these tests so they run in CI without embedding files. Heuristic gives deterministic routing — `task=reasoning` → 122B, `task=graph_indexing` → 122B, `model=35b` override → 35B.

3. **Port choices.** `port122b = 1` causes immediate `ECONNREFUSED`. If port 1 is privileged on macOS, pick another always-unused port like `port122b = 65530`. Actually safest: bind a fake on a random port, capture the port, then immediately Dispose so subsequent connection attempts fail. **Use the latter pattern** (start fake → capture port → stop fake → use port for "down" upstream).

4. **DecisionLog timing.** The decision logger is a Channel-backed BackgroundService. After a request completes, `Thread.Sleep(500)` gives the writer time to flush. If flakiness emerges, increase to 1500ms or use the existing graceful-drain helper.

5. **`testSequenced` is mandatory.** All these tests bind real ports — running in parallel triggers port contention.

6. **Test count budget.** 5 new tests minimum. If startTestRouter helper extraction itself yields a 6th unit test ("/health JSON parses", "pre-flight helper unit"), add it in the same file.
